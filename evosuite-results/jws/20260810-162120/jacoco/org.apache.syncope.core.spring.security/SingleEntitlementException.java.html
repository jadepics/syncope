package org.apache.syncope.core.spring.security;

public class SingleEntitlementException extends RuntimeException {

    private static final long serialVersionUID = 1608618051467782529L;

    private final String entitlement;

    public SingleEntitlementException(final String entitlement) {
        this.entitlement = entitlement;
    }

    public String getEntitlement() {
        return entitlement;
    }
}
